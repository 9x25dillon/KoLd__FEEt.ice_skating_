cmake_minimum_required(VERSION 3.20)
project(kold_native_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(kold_protocol STATIC src/KoLdProtocol.cpp src/KoLdMath.cpp)
target_include_directories(kold_protocol PUBLIC include reference)
add_executable(native_check tests/native_check.cpp)
target_link_libraries(native_check PRIVATE kold_protocol)
foreach(target kold_protocol native_check)
  if(MSVC)
    target_compile_options(${target} PRIVATE /fp:strict /W4)
  else()
    target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME protocol_reference COMMAND native_check "${CMAKE_CURRENT_SOURCE_DIR}/reference")
